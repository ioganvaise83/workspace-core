# Universal Mail MCP — configuration reference. NO SECRETS IN THIS FILE.
#
# This is documentation, not a file the container reads. Every variable already
# has a working default in compose.yaml, and the defaults start the service with
# ZERO accounts, which is a valid state.
#
# Layering, outermost first:
#   .env  (host, git-ignored)  ->  compose.yaml interpolation  ->  container env
#
# The variables below are for the `mail-imap-mcp` service. They go in the
# repository's `.env` on the host, which is git-ignored.

# --- Service wiring --------------------------------------------------------

# The internal container port is fixed at 8765 in compose.yaml and is NOT
# configurable through `.env`: the registry entry, the healthcheck and the
# container all name that port, and a mismatch would silently unregister the
# integration.
#
# The port is never published to the host; the harness reaches the service over
# the compose network at http://mail-imap-mcp:8765/mcp.

# Image tag for the built container.
# MAIL_IMAP_MCP_IMAGE=personal-ai-workspace/mail-imap-mcp:0.2.0

# --- Where the accounts live (host paths, bind-mounted read-only) ----------
#
# accounts.json is the account LIST: ids, providers, logins, auth type, folder
# roles, whether writes are allowed. It holds no passwords and no tokens.
# Start from accounts.example.json.
#
# MAIL_IMAP_CONFIG_DIR=./runtime/mail-imap
#
# One app-password file per password account, named exactly as the account's
# `passwordFile` says. Files should be mode 600 on the host; the container reads
# them as uid 1000, so on Linux they must be readable by that user.
#
# MAIL_IMAP_SECRETS_DIR=./runtime/mail-imap/secrets

# --- Where previews, approvals and the journal live -----------------------
#
# This directory is mounted READ-WRITE into mail-imap-mcp and deliberately NOT
# mounted into the harness. That asymmetry is the whole approval mechanism: the
# agent can read a digest and ask for approval, but cannot write the approval
# file itself. Do not "helpfully" expose it to the harness.
#
# MAIL_IMAP_STATE_DIR=./runtime/mail-imap/state
#
# The approval script writes to ./runtime/mail-imap/state/approvals by default.
# Set the same path if you move it:
#   sh scripts/mail-approve.sh <digest>
# MAIL_APPROVAL_DIR=./runtime/mail-imap/state/approvals

# --- Timing ----------------------------------------------------------------
#
# Per-tool-call ceiling. A stuck or very slow IMAP server is cut off here.
# MAIL_TOOL_TIMEOUT_MS=60000
#
# How long a preview stays valid before it must be re-run.
# MAIL_PREVIEW_TTL_MS=900000
#
# How long an owner approval stays valid.
# MAIL_APPROVAL_TTL_MS=600000

# --- TLS -------------------------------------------------------------------
#
# No variable disables certificate verification, by design. To trust an extra
# root (a corporate TLS-inspecting proxy), point CUSTOM_CA_PATH at a PEM on the
# host. It is mounted read-only and used for BOTH the build (npm ci) and the
# IMAP connections. The system trust store is still honoured.
#
# CUSTOM_CA_PATH=C:/Users/you/kaspersky-ca.pem

# --- Gmail IMAP OAuth ------------------------------------------------------
#
# Not an environment variable: the two paths live in accounts.json under
# `oauth.clientFile` and `oauth.tokenFile`. They are:
#
#   clientFile  Google "installed app" client JSON (client_id + client_secret),
#               created once in the Google Cloud console
#   tokenFile   refresh-token document written by the one-time host
#               authorisation (see the integration README)
#
# They MUST be separate from the existing Gmail API integration's files under
# ~/.gmail-mcp. That token has the gmail.readonly scope, which cannot
# authenticate IMAP, and overwriting it would break the working Gmail MCP.

# --- The bridge token ------------------------------------------------------
#
# Intentionally NOT configurable here. The `mail-token` service mints it onto the
# dsh-home volume, and both the harness bootstrap and this container read it from
# there.
