# Optional corporate CA certificate (TLS inspection: Kaspersky Endpoint
# Security, Zscaler, an HTTPS-inspecting proxy).
#
# Building gmail-mcp runs `npm install` over HTTPS, so a proxy that re-signs TLS
# makes the build fail with SELF_SIGNED_CERT_IN_CHAIN. Point this at the
# inspecting root certificate in PEM form and the build, plus the running
# containers, will trust it in addition to the normal roots. TLS verification is
# NOT disabled.
#
# Export the certificate on the host, keep it outside this repository, and use
# forward slashes on Windows so neither YAML nor the shell eats a backslash:
#
#   Windows: CUSTOM_CA_PATH=C:/Users/you/kaspersky-ca.pem
#   Linux:   CUSTOM_CA_PATH=/home/you/certs/corporate-ca.pem
#
# The mount is read-only; on Linux make the file world-readable (`chmod 644`)
# because the containers do not run as your uid.
# Leave this commented out when you have no TLS inspection: the default is a
# zero-byte placeholder, so an empty `.env` stays valid.
# CUSTOM_CA_PATH=./integrations/ca/empty-ca.pem

# --- Gmail MCP (built-in integration) --------------------------------------

#

GMAIL_MCP_AUTH_DIR=./runtime/gmail/oauth

GMAIL_MCP_IMAGE=personal-ai-workspace/gmail-mcp:3.0.0

# The MCP bridge token is intentionally NOT configured here: the `mcp-token`

# file you edit or commit.

# --- IMAP MCP (built-in integration) ---------------------------------------

# Host directory holding `accounts.json` and a `secrets/` directory with one
# application-password file per mailbox. Mounted read-only into mail-imap-mcp.
# Both are git-ignored (they live under runtime/).
#
# The service starts fine when this directory is empty or missing: it then reports
# zero accounts, which is a valid state. Start from
# integrations/mail-imap/accounts.example.json.
MAIL_IMAP_CONFIG_DIR=./runtime/mail-imap

# The secrets directory may be separate from the accounts file, so the passwords
# can live somewhere with stricter host permissions. One file per account, named
# exactly as the account's `passwordFile` says.
MAIL_IMAP_SECRETS_DIR=./runtime/mail-imap/secrets

# Where previews, owner approvals and the operation journal live. Mounted
# read-write into mail-imap-mcp and deliberately NOT into the harness: that is
# what makes an approval something the agent cannot produce for itself.
MAIL_IMAP_STATE_DIR=./runtime/mail-imap/state

# Image tag for the built IMAP MCP container.
MAIL_IMAP_MCP_IMAGE=personal-ai-workspace/mail-imap-mcp:0.1.0
